### Separación de Datos

| Tipo | Propósito | Entorno | Ejecución |
|------|-----------|---------|-----------|
| **Seed de Producción** | Datos esenciales para que el sistema funcione | Todos | Siempre |

| **Seed de Testing** | Datos específicos para tests automatizados | Test | Antes de tests |

## 5. Ejecución del Seed

### Archivo Principal

```typescript
// prisma/seed.ts

import { PrismaClient } from '@prisma/client';
import { categoriesData } from './seeds/production/categories.seed';
import { adminUser } from './seeds/production/admin.seed';
import { placesData, placePhotosData } from './seeds/development/places.seed';
import { placeHoursData } from './seeds/development/hours.seed';
import { eventsData } from './seeds/development/events.seed';
import { promotionsData } from './seeds/development/promotions.seed';
import { usersData } from './seeds/development/users.seed';
import { reviewsData } from './seeds/development/reviews.seed';

const prisma = new PrismaClient();

async function main() {
  const isProduction = process.env.NODE_ENV === 'production';
  const seedDevelopment = process.env.SEED_DEVELOPMENT === 'true';

  console.log('🌱 Iniciando seed de base de datos...');

  // 1. Categorías (siempre)
  console.log('📦 Insertando categorías...');
  const categories = await Promise.all(
    categoriesData.map((cat) =>
      prisma.category.upsert({
        where: { slug: cat.slug },
        update: cat,
        create: cat,
      })
    )
  );
  console.log(`✅ ${categories.length} categorías insertadas`);

  // 2. Usuario admin (siempre)
  console.log('👤 Creando usuario administrador...');
  await prisma.user.upsert({
    where: { email: adminUser.email },
    update: adminUser,
    create: adminUser,
  });
  console.log('✅ Usuario admin creado');

  // 3. Datos de desarrollo (opcional)
  if (!isProduction || seedDevelopment) {
    console.log('🏗️  Insertando datos de desarrollo...');

    // Usuarios
    const users = await Promise.all(
      usersData.map((user) =>
        prisma.user.upsert({
          where: { email: user.email },
          update: user,
          create: user,
        })
      )
    );
    console.log(`✅ ${users.length} usuarios de ejemplo`);

    // Lugares
    const categoryMap = new Map(categories.map((c) => [c.slug, c.id]));
    
    for (const place of placesData) {
      const { categorySlug, ...placeData } = place;
      const createdPlace = await prisma.place.upsert({
        where: { id: placeData.name.toLowerCase().replace(/\s+/g, '-') },
        update: { ...placeData, categoryId: categoryMap.get(categorySlug)! },
        create: { ...placeData, categoryId: categoryMap.get(categorySlug)! },
      });

      // Fotos del lugar
      const photos = placePhotosData.filter((p) => p.placeSlug === placeData.name.toLowerCase().replace(/\s+/g, '-'));
      for (const photo of photos) {
        await prisma.placePhoto.create({
          data: {
            placeId: createdPlace.id,
            url: photo.url,
            displayOrder: photo.displayOrder,
          },
        });
      }

      // Horarios del lugar
      const hours = placeHoursData.filter((h) => h.placeSlug === placeData.name.toLowerCase().replace(/\s+/g, '-'));
      for (const hour of hours) {
        const { placeSlug, ...hourData } = hour;
        await prisma.placeHour.create({
          data: { ...hourData, placeId: createdPlace.id },
        });
      }
    }
    console.log(`✅ ${placesData.length} lugares de ejemplo`);

    // Eventos
    await prisma.event.createMany({ data: eventsData });
    console.log(`✅ ${eventsData.length} eventos de ejemplo`);

    // Promociones
    for (const promo of promotionsData) {
      const { placeSlug, ...promoData } = promo;
      const place = await prisma.place.findFirst({
        where: { name: placesData.find((p) => p.name.toLowerCase().replace(/\s+/g, '-') === placeSlug)?.name },
      });
      if (place) {
        await prisma.promotion.create({
          data: { ...promoData, placeId: place.id },
        });
      }
    }
    console.log(`✅ ${promotionsData.length} promociones de ejemplo`);

    // Reseñas
    for (const review of reviewsData) {
      const { userIndex, placeSlug, ...reviewData } = review;
      const user = users[userIndex];
      const place = await prisma.place.findFirst({
        where: { name: placesData.find((p) => p.name.toLowerCase().replace(/\s+/g, '-') === placeSlug)?.name },
      });
      if (user && place) {
        await prisma.review.create({
          data: { ...reviewData, userId: user.id, placeId: place.id },
        });
      }
    }
    console.log(`✅ ${reviewsData.length} reseñas de ejemplo`);
  }

  console.log('🎉 Seed completado exitosamente!');
}

main()
  .catch((e) => {
    console.error('❌ Error en seed:', e);
    process.exit(1);
  })
  .finally(async () => {
    await prisma.$disconnect();
  });
```
