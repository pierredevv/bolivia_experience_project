## 2. Funciones Auxiliares

```sql
-- Función para actualizar updated_at automáticamente
CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = NOW();
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Función para recalcular rating_avg y rating_count de un lugar
CREATE OR REPLACE FUNCTION recalculate_place_rating()
RETURNS TRIGGER AS $$
BEGIN
    UPDATE places
    SET 
        rating_avg = COALESCE((
            SELECT AVG(rating)::DECIMAL(3,2)
            FROM reviews
            WHERE place_id = COALESCE(NEW.place_id, OLD.place_id)
            AND is_approved = true
        ), 0),
        rating_count = COALESCE((
            SELECT COUNT(*)::INTEGER
            FROM reviews
            WHERE place_id = COALESCE(NEW.place_id, OLD.place_id)
            AND is_approved = true
        ), 0)
    WHERE id = COALESCE(NEW.place_id, OLD.place_id);
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Función para normalizar slugs (sin acentos, lowercase, guiones)
CREATE OR REPLACE FUNCTION generate_slug(input_text TEXT)
RETURNS TEXT AS $$
BEGIN
    RETURN lower(regexp_replace(
        regexp_replace(
            unaccent(input_text),
            '[^a-zA-Z0-9\s]', '', 'g'
        ),
        '\s+', '-', 'g'
    ));
END;
$$ LANGUAGE plpgsql IMMUTABLE;
```

## 3. Tablas

### 3.1 users

```sql
CREATE TABLE users (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    email VARCHAR(255) UNIQUE NOT NULL,
    name VARCHAR(255) NOT NULL,
    photo_url TEXT,
    country VARCHAR(100),
    language VARCHAR(5) NOT NULL DEFAULT 'es'
        CHECK (language IN ('es', 'en', 'pt')),
    role VARCHAR(20) NOT NULL DEFAULT 'usuario'
        CHECK (role IN ('admin', 'empresa', 'usuario')),
    firebase_uid VARCHAR(255) UNIQUE,
    is_active BOOLEAN NOT NULL DEFAULT true,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- Índices
CREATE INDEX idx_users_email ON users(email);
CREATE INDEX idx_users_firebase_uid ON users(firebase_uid);
CREATE INDEX idx_users_role ON users(role);
CREATE INDEX idx_users_active ON users(is_active) WHERE is_active = true;

-- Trigger updated_at
CREATE TRIGGER trg_users_updated_at
    BEFORE UPDATE ON users
    FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();
```

### 3.3 places

```sql
CREATE TABLE places (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(255) NOT NULL,
    description TEXT,
    description_en TEXT,
    address VARCHAR(500) NOT NULL,
    phone VARCHAR(50),
    website VARCHAR(255),
    instagram VARCHAR(255),
    facebook VARCHAR(255),
    tiktok VARCHAR(255),
    latitude DECIMAL(10, 8) NOT NULL,
    longitude DECIMAL(11, 8) NOT NULL,
    location GEOGRAPHY(POINT, 4326) GENERATED ALWAYS AS (
        ST_SetSRID(ST_MakePoint(longitude, latitude), 4326)
    ) STORED,
    rating_avg DECIMAL(3, 2) NOT NULL DEFAULT 0.00
        CHECK (rating_avg >= 0 AND rating_avg <= 5),
    rating_count INTEGER NOT NULL DEFAULT 0
        CHECK (rating_count >= 0),
    category_id UUID NOT NULL REFERENCES categories(id) ON DELETE RESTRICT,
    owner_id UUID REFERENCES users(id) ON DELETE SET NULL,
    is_featured BOOLEAN NOT NULL DEFAULT false,
    is_active BOOLEAN NOT NULL DEFAULT true,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- Índices
CREATE INDEX idx_places_category ON places(category_id);
CREATE INDEX idx_places_owner ON places(owner_id);
CREATE INDEX idx_places_location ON places USING GIST(location);
CREATE INDEX idx_places_rating ON places(rating_avg DESC);
CREATE INDEX idx_places_featured ON places(is_featured) WHERE is_featured = true;
CREATE INDEX idx_places_active ON places(is_active) WHERE is_active = true;
CREATE INDEX idx_places_name_trgm ON places USING GIN(name gin_trgm_ops);

-- Trigger updated_at
```

### 3.6 reviews

```sql
CREATE TABLE reviews (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    place_id UUID NOT NULL REFERENCES places(id) ON DELETE CASCADE,
    rating SMALLINT NOT NULL CHECK (rating >= 1 AND rating <= 5),
    comment TEXT,
    photos TEXT[],
    visit_date DATE,
    is_approved BOOLEAN NOT NULL DEFAULT false,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    
    -- Un usuario solo puede tener 1 review por lugar
    UNIQUE(user_id, place_id)
);

-- Índices
CREATE INDEX idx_reviews_place ON reviews(place_id);
CREATE INDEX idx_reviews_user ON reviews(user_id);
CREATE INDEX idx_reviews_rating ON reviews(rating);
CREATE INDEX idx_reviews_approved ON reviews(is_approved) WHERE is_approved = true;
CREATE INDEX idx_reviews_created ON reviews(created_at DESC);

-- Trigger updated_at
CREATE TRIGGER trg_reviews_updated_at
    BEFORE UPDATE ON reviews
    FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

-- Trigger para recalcular rating del lugar
CREATE TRIGGER trg_reviews_recalculate_insert
    AFTER INSERT ON reviews
    FOR EACH ROW EXECUTE FUNCTION recalculate_place_rating();

CREATE TRIGGER trg_reviews_recalculate_update
    AFTER UPDATE OF is_approved, rating ON reviews
    FOR EACH ROW EXECUTE FUNCTION recalculate_place_rating();

CREATE TRIGGER trg_reviews_recalculate_delete
    AFTER DELETE ON reviews
    FOR EACH ROW EXECUTE FUNCTION recalculate_place_rating();
```

## 4. Vistas Materializadas

```sql
-- Vista materializada: Lugares populares (actualizar cada hora)
CREATE MATERIALIZED VIEW mv_popular_places AS
SELECT 
    p.id,
    p.name,
    p.description,
    p.address,
    p.latitude,
    p.longitude,
    p.rating_avg,
    p.rating_count,
    p.is_featured,
    c.name as category_name,
    c.icon as category_icon,
    (
        SELECT url FROM place_photos 
        WHERE place_id = p.id 
        ORDER BY display_order 
        LIMIT 1
    ) as primary_photo
FROM places p
JOIN categories c ON p.category_id = c.id
WHERE p.is_active = true
ORDER BY p.is_featured DESC, p.rating_avg DESC, p.rating_count DESC;

CREATE UNIQUE INDEX idx_mv_popular_places_id ON mv_popular_places(id);

-- Vista materializada: Estadísticas de categorías
CREATE MATERIALIZED VIEW mv_category_stats AS
SELECT 
    c.id,
    c.name,
    c.name_en,
    c.icon,
    c.slug,
    COUNT(p.id) as place_count,
    COALESCE(AVG(p.rating_avg), 0)::DECIMAL(3,2) as avg_rating
FROM categories c
LEFT JOIN places p ON p.category_id = c.id AND p.is_active = true
WHERE c.is_active = true
GROUP BY c.id, c.name, c.name_en, c.icon, c.slug;

CREATE UNIQUE INDEX idx_mv_category_stats_id ON mv_category_stats(id);
```

## 5. Políticas de Seguridad a Nivel de Fila (RLS)

```sql
-- Habilitar RLS en tablas sensibles
ALTER TABLE users ENABLE ROW LEVEL SECURITY;
ALTER TABLE reviews ENABLE ROW LEVEL SECURITY;
ALTER TABLE favorites ENABLE ROW LEVEL SECURITY;
ALTER TABLE notifications ENABLE ROW LEVEL SECURITY;
ALTER TABLE search_history ENABLE ROW LEVEL SECURITY;

-- Política: Usuarios solo pueden ver/editar su propio perfil
CREATE POLICY users_own_profile ON users
    FOR ALL
    USING (id = current_setting('app.current_user_id')::UUID)
    WITH CHECK (id = current_setting('app.current_user_id')::UUID);

-- Política: Reviews aprobadas son públicas, propias siempre visibles
CREATE POLICY reviews_public ON reviews
    FOR SELECT
    USING (is_approved = true OR user_id = current_setting('app.current_user_id')::UUID);

-- Política: Usuarios solo pueden crear/editar/eliminar sus propias reviews
CREATE POLICY reviews_own ON reviews
    FOR ALL
    USING (user_id = current_setting('app.current_user_id')::UUID)
    WITH CHECK (user_id = current_setting('app.current_user_id')::UUID);

-- Política: Favoritos solo del usuario actual
CREATE POLICY favorites_own ON favorites
    FOR ALL
    USING (user_id = current_setting('app.current_user_id')::UUID)
    WITH CHECK (user_id = current_setting('app.current_user_id')::UUID);

-- Política: Notificaciones solo del usuario actual
CREATE POLICY notifications_own ON notifications
    FOR ALL
    USING (user_id = current_setting('app.current_user_id')::UUID);

-- Política: Historial de búsqueda solo del usuario actual
CREATE POLICY search_history_own ON search_history
    FOR ALL
    USING (user_id = current_setting('app.current_user_id')::UUID OR user_id IS NULL);
```
